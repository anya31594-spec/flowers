function addTouchSparks(x, y) { for (let i = 0; i < 10; i++) { touchParticles.push({ x: x, y: y, vx: (Math.random() - 0.5) * 5, vy: (Math.random() - 0.5) * 5, size: 2 + Math.random() * 3, life: 1 }); } } function animate() { time += 1; ctx.clearRect(0, 0, width, height); // Фон із м'яким градієнтом let bgGlow = ctx.createRadialGradient(width / 2, height, 50, width / 2, height / 2, width); bgGlow.addColorStop(0, '#210b36'); bgGlow.addColorStop(1, '#08020f'); ctx.fillStyle = bgGlow; ctx.fillRect(0, 0, width, height); // Світлячки particles.forEach(p => { p.update(); p.draw(); }); // Квіти flowers.forEach(f => { f.update(); f.draw(time); }); // Ефекти при торканні for (let i = touchParticles.length - 1; i >= 0; i--) { let p = touchParticles[i]; p.x += p.vx; p.y += p.vy; p.life -= 0.025; if (p.life <= 0) { touchParticles.splice(i, 1); continue; } ctx.save(); ctx.globalAlpha = p.life; ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2); ctx.fillStyle = '#ff70a6'; ctx.shadowColor = '#ff4d6d'; ctx.shadowBlur = 8; ctx.fill(); ctx.restore(); } requestAnimationFrame(animate); } window.addEventListener('resize', resize); window.addEventListener('touchstart', (e) => { for (let touch of e.touches) { addTouchSparks(touch.clientX, touch.clientY); } }); window.addEventListener('mousedown', (e) => { addTouchSparks(e.clientX, e.clientY); }); resize(); animate();
